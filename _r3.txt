<script>
window.__probe = { errs: [], steps: [] };
window.addEventListener('error', (e) => window.__probe.errs.push('global: ' + e.message));
window.addEventListener('unhandledrejection', (e) => window.__probe.errs.push('reject: ' + e.reason));
</script>
<script type="module">
import './src/main.js';

const out = window.__probe;
const step = (name, fn) => {
  try {
    const r = fn();
    out.steps.push(name + ':' + (r === undefined ? 'ok' : r));
  } catch (e) {
    out.steps.push(name + ':THROW ' + (e && e.message));
    out.errs.push(name + ': ' + (e && e.stack ? e.stack.split('\n').slice(0, 3).join(' | ') : e));
  }
};
const tick = (g, n) => { for (let i = 0; i < n; i++) g.update(1 / 60); };
const hud = (g) => { g.hudStamp = 0; g.updateHud(); };

function mainProbe() {
  const g = window.godkiller;
  const d = g.dom;

  step('hubPips', () => {
    hud(g);
    return 'hearts=' + (d.hearts.innerHTML.match(/<i/g) || []).length
      + ' armorPips=' + (d.armor.innerHTML.match(/<i/g) || []).length
      + ' modChipHidden=' + d.modChip.classList.contains('hidden');
  });

  step('upgradePips', () => {
    g.save.credits = 999999;
    g.save.hearts = 3;
    g.save.armor = 2;
    hud(g);
    return 'hearts=' + (d.hearts.innerHTML.match(/<i/g) || []).length
      + ' armorPips=' + (d.armor.innerHTML.match(/<i/g) || []).length;
  });

  step('modChipMission', () => {
    g.save.accepted = g.save.contracts[0];
    g.startMission();
    hud(g);
    tick(g, 2);
    hud(g);
    return 'hidden=' + d.modChip.classList.contains('hidden') + ' len=' + d.modChip.innerHTML.length
      + ' perks=' + (d.perkbar.innerHTML.match(/class="perk/g) || []).length;
  });

  step('hintBox', () => {
    const h = d.hintBox.innerHTML || d.hintBox.textContent;
    return 'hint=' + h.replace(/<[^>]+>/g, '').slice(0, 40);
  });

  step('comboHud', () => {
    g.combo = 5;
    g.hudStamp = 0;
    g.updateHud();
    return 'text=' + d.combo.textContent + ' on=' + d.combo.classList.contains('on');
  });

  step('longRun', () => {
    tick(g, 3600);
    const p = g.player;
    const bad = !isFinite(p.x) || !isFinite(p.y) || !isFinite(g.cam.x) || !isFinite(g.cam.y);
    return 't=' + g.missionTime.toFixed(1) + ' hp=' + p.hp + ' alive=' + p.alive
      + ' nan=' + bad + ' state=' + g.state;
  });

  step('deadHud', () => {
    if (g.player.alive) g.player.hit(g, { label: 'PROBE' });
    g.openResults('dead', 0, false, 'PROBE');
    hud(g);
    const html = d.hearts.innerHTML;
    return 'hp=' + g.player.hp + ' offPips=' + (html.match(/off/g) || []).length + ' hitPips=' + (html.match(/hit/g) || []).length;
  });

  step('singleExtract', () => {
    g.state = 'MISSION';
    g.player.alive = true;
    g.player.hp = 1;
    const enemiesBefore = g.enemies.length;
    for (const e of g.enemies.slice()) {
      if (e.isBoss) continue;
      e.dead = true;
    }
    tick(g, 1);
    const t0 = g.save.trophies.length;
    g.player.x = g.map.pad.x; g.player.y = g.map.pad.y;
    g.tryExtract();
    g.tryExtract();
    const t1 = g.save.trophies.length;
    return 'enemies=' + enemiesBefore + ' trophies+' + (t1 - t0) + ' state=' + g.state;
  });

  step('renderWin', () => { g.render(); return 'ok'; });
  step('backHub', () => { g.hideOverlay(); tick(g, 60); g.render(); return g.state; });
}

if (window.godkiller) mainProbe();
else out.errs.push('no window.godkiller');
document.title = 'RESULT ' + JSON.stringify(out);
</script>
