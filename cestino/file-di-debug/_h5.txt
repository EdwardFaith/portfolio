<script type="module">
import './src/main.js';

const out = window.__probe;
const step = (name, fn) => {
  try {
    const r = fn();
    out.steps.push(name + ':' + (r === undefined ? 'ok' : r));
  } catch (e) {
    out.steps.push(name + ':THROW ' + (e && e.message));
    out.errs.push(name + ': ' + (e && e.stack ? e.stack.split('\n').slice(0, 3).join(' | ') : e));
  }
};
const withTimeout = (p, ms) => Promise.race([p, new Promise((r) => setTimeout(() => r('timeout'), ms))]);

async function main() {
  const g = window.godkiller;
  step('boot', () => (g ? 'game ok' : 'NO GAME'));
  if (!g) { document.title = 'RESULT ' + JSON.stringify(out); return; }

  step('t0', () => g.t.toFixed(2));
  const raf = await withTimeout(new Promise((r) => requestAnimationFrame(() => r('raf'))), 600);
  out.steps.push('rafOnce:' + raf);
  await withTimeout(new Promise((r) => setTimeout(r, 1500)), 4000);
  step('t1', () => g.t.toFixed(2) + ' fade=' + g.fade.toFixed(2) + ' acc=' + g.acc.toFixed(3));
  step('hudHearts', () => d().hearts.innerHTML.length + ' pips=' + (d().hearts.innerHTML.match(/<i/g) || []).length);
  step('hudArmor', () => d().armor.innerHTML.length + ' pips=' + (d().armor.innerHTML.match(/<i/g) || []).length);
  step('hudCredits', () => JSON.stringify(d().credits.textContent));
  step('hudObj', () => d().objName.textContent + ' / ' + d().objSub.textContent);
  step('hudPerks', () => d().perkbar.innerHTML.length + ' hint=' + JSON.stringify(d().hintBox.textContent.slice(0, 42)));
  step('loadingClass', () => JSON.stringify(d().loading.className));
  step('canvasCenter', () => Array.from(g.ctx.getImageData(240, 135, 1, 1).data).join(','));
  step('manualFrames', () => {
    const t = g.t;
    for (let i = 0; i < 30; i++) g.frame(performance.now() + i * 16.7);
    return 'dt=' + (g.t - t).toFixed(2) + ' state=' + g.state;
  });
  step('startMissionViaHub', () => {
    g.save.accepted = g.save.contracts[0];
    g.startMission();
    for (let i = 0; i < 120; i++) g.update(1 / 60);
    g.render();
    return g.state + ' enemies=' + g.enemies.length + ' hearts=' + d().hearts.innerHTML.length;
  });
  document.title = 'RESULT ' + JSON.stringify(out);
}

function d() { return window.godkiller.dom; }

main().catch((e) => {
  out.errs.push('main: ' + (e && e.message));
  document.title = 'RESULT ' + JSON.stringify(out);
});
